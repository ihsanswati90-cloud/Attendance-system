// Load saved attendance function loadData() { let data = JSON.parse(localStorage.getItem("attendance")) || []; let list = document.getElementById("list"); list.innerHTML = ""; data.forEach(name => { let li = document.createElement("li"); li.innerText = name + " - Present"; list.appendChild(li); }); } // Mark attendance function markAttendance() { let name = document.getElementById("name").value; if (name === "") { alert("Enter name"); return; } let data = JSON.parse(localStorage.getItem("attendance")) || []; data.push(name); localStorage.setItem("attendance", JSON.stringify(data)); document.getElementById("name").value = ""; loadData(); } // Show current date window.onload = function() { document.getElementById("date").innerText = new Date().toDateString(); loadData(); }
